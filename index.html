Rode depois do 003. -- ===================================================================== -- Resumo do período: totais por hora, dia ou mês e os outros itens vendidos. -- p_gran: 'hour', 'day' ou 'month' (horário de Fortaleza). create or replace function public.financeiro_resumo(p_inicio timestamptz, p_fim timestamptz, p_gran text default 'day') returns jsonb language plpgsql stable security definer set search_path = public as $$ declare r jsonb; begin if not public.eh_admin() then raise exception 'Apenas administradores acessam o financeiro.'; end if; if p_gran is null or p_gran not in ('hour', 'day', 'month') then raise exception 'Agrupamento inválido.'; end if; with v as ( select id, date_trunc(p_gran, criado_em at time zone 'America/Fortaleza') as b, total_centavos from public.vendas where criado_em >= p_inicio and criado_em < p_fim ), i as ( select vi.venda_id, coalesce(sum(vi.gramas) filter (where vi.grupo = 'comida'), 0) as gc, coalesce(sum(vi.gramas) filter (where vi.grupo = 'acai'), 0) as ga, count(*) filter (where vi.grupo = 'produto') as it from public.venda_itens vi join v on v.id = vi.venda_id group by vi.venda_id ), p as ( select vp.venda_id, coalesce(sum(vp.valor_centavos) filter (where vp.forma = 'pix'), 0) as pix, coalesce(sum(vp.valor_centavos) filter (where vp.forma = 'credito'), 0) as credito, coalesce(sum(vp.valor_centavos) filter (where vp.forma = 'debito'), 0) as debito, coalesce(sum(vp.valor_centavos) filter (where vp.forma = 'dinheiro'), 0) as dinheiro from public.venda_pagamentos vp join v on v.id = vp.venda_id group by vp.venda_id ), bk as ( select v.b, count(*) as n, sum(v.total_centavos) as fat, sum(coalesce(i.gc, 0)) as gc, sum(coalesce(i.ga, 0)) as ga, sum(coalesce(i.it, 0)) as it, sum(coalesce(p.pix, 0)) as pix, sum(coalesce(p.credito, 0)) as credito, sum(coalesce(p.debito, 0)) as debito, sum(coalesce(p.dinheiro, 0)) as dinheiro from v left join i on i.venda_id = v.id left join p on p.venda_id = v.id group by v.b ), pr as ( select vi.descricao as nome, count(*) as qtd, sum(vi.valor_centavos) as valor from public.venda_itens vi join v on v.id = vi.venda_id where vi.grupo = 'produto' group by vi.descricao ) select jsonb_build_object( 'buckets', coalesce((select jsonb_agg(jsonb_build_object( 'k', to_char(b, 'YYYY-MM-DD"T"HH24'), 'n', n, 'fat', fat, 'gCom', gc, 'gAcai', ga, 'itens', it, 'pix', pix, 'credito', credito, 'debito', debito, 'dinheiro', dinheiro) order by b) from bk), '[]'::jsonb), 'produtos', coalesce((select jsonb_agg(jsonb_build_object('nome', nome, 'qtd', qtd, 'valor', valor) order by qtd desc, nome) from pr), '[]'::jsonb)) into r; return r; end $$; -- Lista de vendas do período, com filtros e paginação. -- Devolve a quantidade e a soma de TODAS as vendas filtradas e uma página de linhas. create or replace function public.financeiro_vendas( p_inicio timestamptz, p_fim timestamptz, p_forma text default null, p_tipo text default null, p_operador uuid default null, p_cliente uuid default null, p_sem_cliente boolean default false, p_limite int default 200, p_pulo int default 0) returns jsonb language plpgsql stable security definer set search_path = public as $$ declare r jsonb; begin if not public.eh_admin() then raise exception 'Apenas administradores acessam o financeiro.'; end if; with f as ( select v.* from public.vendas v where v.criado_em >= p_inicio and v.criado_em < p_fim and (p_tipo is null or v.tipo = p_tipo) and (p_operador is null or v.operador_id = p_operador) and (p_cliente is null or v.cliente_id = p_cliente) and (not coalesce(p_sem_cliente, false) or v.cliente_id is null) and (p_forma is null or exists (select 1 from public.venda_pagamentos x where x.venda_id = v.id and x.forma = p_forma)) ), pg as ( select * from f order by criado_em desc, numero desc limit greatest(1, least(coalesce(p_limite, 200), 1000)) offset greatest(0, coalesce(p_pulo, 0)) ) select jsonb_build_object( 'qtd', (select count(*) from f), 'soma', (select coalesce(sum(total_centavos), 0) from f), 'linhas', coalesce((select jsonb_agg(jsonb_build_object( 'id', pg.id, 'numero', pg.numero, 'criado_em', pg.criado_em, 'operador_id', pg.operador_id, 'operador', (select nome from public.perfis where id = pg.operador_id), 'cliente_id', pg.cliente_id, 'cliente', (select nome from public.clientes where id = pg.cliente_id), 'tipo', pg.tipo, 'total', pg.total_centavos, 'desconto', pg.desconto_centavos, 'itens', coalesce((select jsonb_agg(jsonb_build_object('descricao', i.descricao, 'gramas', i.gramas, 'valor', i.valor_centavos) order by i.grupo, i.descricao) from public.venda_itens i where i.venda_id = pg.id), '[]'::jsonb), 'pagamentos', coalesce((select jsonb_agg(jsonb_build_object('rotulo', x.rotulo, 'forma', x.forma, 'valor', x.valor_centavos) order by x.ordem) from public.venda_pagamentos x where x.venda_id = pg.id), '[]'::jsonb) ) order by pg.criado_em desc, pg.numero desc) from pg), '[]'::jsonb)) into r; return r; end $$; -- Fluxos de caixa abertos no período (e os que ainda estão abertos), com os totais de cada um. create or replace function public.financeiro_fluxos(p_inicio timestamptz, p_fim timestamptz) returns jsonb language plpgsql stable security definer set search_path = public as $$ declare r jsonb; begin if not public.eh_admin() then raise exception 'Apenas administradores acessam o financeiro.'; end if; select coalesce(jsonb_agg(jsonb_build_object( 'id', f.id, 'operador_id', f.operador_id, 'operador', (select nome from public.perfis where id = f.operador_id), 'numero', f.numero, 'dia', f.dia, 'aberto_em', f.aberto_em, 'fechado_em', f.fechado_em, 'fundo', f.fundo_centavos, 'contado', f.contado_centavos, 'n', (select count(*) from public.vendas v where v.fluxo_id = f.id), 'total', (select coalesce(sum(v.total_centavos), 0) from public.vendas v where v.fluxo_id = f.id), 'pix', (select coalesce(sum(x.valor_centavos), 0) from public.venda_pagamentos x join public.vendas v on v.id = x.venda_id where v.fluxo_id = f.id and x.forma = 'pix'), 'credito', (select coalesce(sum(x.valor_centavos), 0) from public.venda_pagamentos x join public.vendas v on v.id = x.venda_id where v.fluxo_id = f.id and x.forma = 'credito'), 'debito', (select coalesce(sum(x.valor_centavos), 0) from public.venda_pagamentos x join public.vendas v on v.id = x.venda_id where v.fluxo_id = f.id and x.forma = 'debito'), 'dinheiro', (select coalesce(sum(x.valor_centavos), 0) from public.venda_pagamentos x join public.vendas v on v.id = x.venda_id where v.fluxo_id = f.id and x.forma = 'dinheiro') ) order by f.aberto_em), '[]'::jsonb) into r from public.fluxos_caixa f where (f.aberto_em >= p_inicio and f.aberto_em < p_fim) or (f.fechado_em is null and f.aberto_em < p_fim); return r; end $$; revoke execute on function public.financeiro_resumo(timestamptz, timestamptz, text), public.financeiro_vendas(timestamptz, timestamptz, text, text, uuid, uuid, boolean, int, int), public.financeiro_fluxos(timestamptz, timestamptz) from public, anon; grant execute on function public.financeiro_resumo(timestamptz, timestamptz, text), public.financeiro_vendas(timestamptz, timestamptz, text, text, uuid, uuid, boolean, int, int), public.financeiro_fluxos(timestamptz, timestamptz) to authenticated; -- --------------------------------------------------------------------- -- Perfil do operador: foto (imagem pequena guardada no próprio perfil) e cor de fundo. -- --------------------------------------------------------------------- alter table public.perfis add column if not exists avatar text; alter table public.perfis add column if not exists cor text; -- O próprio usuário altera nome, foto e cor. Perfil (admin/caixa) e acesso continuam só com o administrador. create or replace function public.atualizar_meu_perfil(p_nome text, p_avatar text default null, p_cor text default null) returns public.perfis language plpgsql security definer set search_path = public as $$ declare v public.perfis; v_nome text := trim(coalesce(p_nome, '')); begin if auth.uid() is null or not public.eh_ativo() then raise exception 'Usuário sem acesso.'; end if; if length(v_nome) < 1 or length(v_nome) > 60 then raise exception 'O nome precisa ter entre 1 e 60 caracteres.'; end if; if p_avatar is not null and (p_avatar not like 'data:image/%' or length(p_avatar) > 200000) then raise exception 'Imagem inválida ou grande demais.'; end if; if p_cor is not null and p_cor !~ '^#[0-9A-Fa-f]{6}$' then raise exception 'Cor inválida.'; end if; update public.perfis set nome = v_nome, avatar = p_avatar, cor = p_cor where id = auth.uid() returning * into v; return v; end $$; revoke execute on function public.atualizar_meu_perfil(text, text, text) from public, anon; grant execute on function public.atualizar_meu_perfil(text, text, text) to authenticated; -- Fim do 005_financeiro_e_perfis.sql
